<p className={styles.exampleQuestion}>How far did AlexNet spread through three generations of citations?</p>
<p className={styles.exampleHow}>BFS follows citations backward from the 2012 paper; SQL counts the papers reached at each hop.</p>

```sql
SELECT b.distance, COUNT(*) AS papers, ROUND(AVG(p.year), 1) AS avg_year
FROM cugraph_bfs(
  edges => (SELECT dst AS src, src AS dst FROM citation_edges_by_dst),
  sources => (SELECT paper_id AS vertex FROM papers
              WHERE title = 'ImageNet Classification with Deep Convolutional Neural Networks'
                AND year = 2012),
  depth_limit => 3, output_mode => 'normalized') b
JOIN papers p ON p.paper_id = b.vertex
WHERE b.reachable
GROUP BY b.distance
ORDER BY b.distance;
```

<p className={styles.exampleQuestion}>Which of 1M catalog products are closest to each of 10,000 others?</p>
<p className={styles.exampleHow}>One call runs 10,000 exact nearest-neighbor searches over text embeddings computed before the query.</p>

```sql
SELECT n.query_id, q.title AS source_title, n.rank,
       n.neighbor_id, p.title AS candidate_title,
       1.0 - n.distance / 2.0 AS cosine_similarity
FROM cuvs_brute_force_knn(
  dataset => (SELECT * FROM product_corpus_1000000),
  queries => (SELECT * FROM product_queries_10000),
  k => 10, metric => 'l2_expanded') n
JOIN semantic_products_text q ON q.id = n.query_id
JOIN semantic_products_text p ON p.id = n.neighbor_id
ORDER BY n.query_id, n.rank;
```

<TabItem value="malware" label="Malware files">

<p className={styles.exampleEyebrow}><span>cuML</span>EMBER2024 challenge split · 6,315 malicious files</p>
<p className={styles.exampleQuestion}>Which files does a pretrained malware model score highest, and why?</p>
<p className={styles.exampleHow}>LightGBM scores every file on the GPU; Top-K ranks SHAP attributions computed on the CPU beforehand.</p>

```sql
SELECT p.id, m.sha256, m.file_type, m.week_id, p.prediction,
       t.rank, g.group_name, t.value AS positive_log_odds_contribution
FROM cuml_forest_predict(
  input => (SELECT id, vector FROM ember_research_vectors_challenge ORDER BY id),
  model => 'EMBER2024_all.model', model_format => 'lightgbm') p
JOIN cuvs_top_k(
  input => (SELECT id, vector FROM ember_attributions ORDER BY id),
  k => 3, select => 'max') t ON t.id = p.id
JOIN ember_feature_groups g ON g.group_index = t.position
JOIN ember_metadata_challenge m ON m.id = p.id
WHERE p.prediction >= 0.99 AND t.value > 0
ORDER BY p.prediction DESC, p.id, t.rank
LIMIT 30;
```

<TabItem value="marketplace" label="Marketplace brands">

<p className={styles.exampleEyebrow}><span>cuGraph</span>MerRec · Mercari US marketplace · 1.27B events · May to October 2023</p>
<p className={styles.exampleQuestion}>Which brands moved closer to Barbie fans after the July 2023 film release?</p>
<p className={styles.exampleHow}>Personalized PageRank restarts at Barbie on the May-June and July-August like graphs; plain PageRank on the same graphs gives each brand's popularity baseline.</p>

```sql
WITH likes AS (
  SELECT e.user_id, i.brand_id,
         CASE WHEN e.event_time < TIMESTAMP '2023-07-01 00:00:00'
              THEN 0 ELSE 1 END AS after_release
  FROM merrec_events e JOIN merrec_items i ON i.item_id = e.item_id
  WHERE e.event_type = 'item_like' AND i.brand_id <> 0
    AND e.event_time >= TIMESTAMP '2023-05-01 00:00:00'
    AND e.event_time < TIMESTAMP '2023-09-01 00:00:00'
  GROUP BY e.user_id, i.brand_id,
           CASE WHEN e.event_time < TIMESTAMP '2023-07-01 00:00:00'
                THEN 0 ELSE 1 END
), before_graph AS (
  SELECT -(user_id + 1) AS src, CAST(brand_id AS BIGINT) AS dst
  FROM likes WHERE after_release = 0
), after_graph AS (
  SELECT -(user_id + 1) AS src, CAST(brand_id AS BIGINT) AS dst
  FROM likes WHERE after_release = 1
), fans AS (
  SELECT CAST(brand_id AS BIGINT) AS brand_id,
         SUM(1 - after_release) AS fans_before, SUM(after_release) AS fans_after
  FROM likes GROUP BY brand_id
), ppr_before AS (
  SELECT vertex, value FROM cugraph_personalized_pagerank(
    edges => (SELECT src, dst FROM before_graph),
    personalization => (SELECT CAST(2595 AS BIGINT) AS vertex, CAST(1.0 AS DOUBLE) AS value),
    directed => false)
), ppr_after AS (
  SELECT vertex, value FROM cugraph_personalized_pagerank(
    edges => (SELECT src, dst FROM after_graph),
    personalization => (SELECT CAST(2595 AS BIGINT) AS vertex, CAST(1.0 AS DOUBLE) AS value),
    directed => false)
), pr_before AS (
  SELECT vertex, value FROM cugraph_pagerank(
    edges => (SELECT src, dst FROM before_graph), directed => false)
), pr_after AS (
  SELECT vertex, value FROM cugraph_pagerank(
    edges => (SELECT src, dst FROM after_graph), directed => false)
), brands AS (
  SELECT CAST(brand_id AS BIGINT) AS brand_id, MIN(brand_name) AS brand_name
  FROM merrec_items WHERE brand_id <> 0 GROUP BY brand_id
)
SELECT b.brand_name, f.fans_before, f.fans_after,
       pb.value / rb.value AS lift_before,
       pa.value / ra.value AS lift_after,
       (pa.value / ra.value) / (pb.value / rb.value) AS lift_change
FROM ppr_after pa
JOIN pr_after ra ON ra.vertex = pa.vertex
JOIN ppr_before pb ON pb.vertex = pa.vertex
JOIN pr_before rb ON rb.vertex = pa.vertex
JOIN fans f ON f.brand_id = pa.vertex
JOIN brands b ON b.brand_id = pa.vertex
WHERE pa.vertex > 0 AND pa.vertex <> 2595
  AND f.fans_before >= 1000 AND f.fans_after >= 1000
ORDER BY lift_change DESC, b.brand_name
LIMIT 25;
```
